2) Настроить протокол динамической маршрутизации RIP v2 для области, указанной на схеме schema-lab5.

настройка R4 роутера:

conf t
int FastEthernet1/0
 ip add 192.168.10.1 255.255.255.0
 no shut
int FastEthernet0/0
 ip add 10.0.45.1 255.255.255.252
 no shut
exit
router rip
 version 2
 no auto-summary
 network 192.168.10.0
 network 10.0.0.0
exit

настройка R5 роутера:

 no shut
exit
router rip

exit

3) Настроить протокол динамической маршрутизации OSPF для зон 0, 1, 2. Зону 1 настроить как полностью (nb!) тупиковую.
4) Настроить редистрибуцию маршрутов между протоколами RIP v2 и OSPF.

настройка R1 роутера:

conf t
int FastEthernet1/0
 ip add 10.0.15.2 255.255.255.252
 no shut
int FastEthernet0/0
 ip add 10.0.123.1 255.255.255.0
 no shut
exit
! Настройка RIP и редистрибуции из OSPF
router rip
 version 2
 no auto-summary
 network 10.0.0.0
 redistribute ospf 1 metric 3
exit
router ospf 1
 network 10.0.123.0 0.0.0.255 area 0
 redistribute rip subnets
exit

В терминале R2
conf t
int FastEthernet0/0
 ip add 10.0.123.2 255.255.255.0
 no shut
int FastEthernet1/0
 ip add 10.0.26.1 255.255.255.252
 no shut
exit
router ospf 1
 network 10.0.123.0 0.0.0.255 area 0
 network 10.0.26.0 0.0.0.3 area 2
exit

В терминале R3

conf t
int FastEthernet0/0

exit

exit
